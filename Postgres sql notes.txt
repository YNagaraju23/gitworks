 Postgres sql notes:

function: 
Example 1:
CREATE OR REPLACE FUNCTION increase(X integer) RETURNS integer
language plpgsql
AS $$
BEGIN
RETURN X+10;
END;
$$;


select increase(5)

Example 2:
CREATE OR REPLACE FUNCTION addition(X integer, Y integer) RETURNS integer
language plpgsql
AS $$
BEGIN
  return x+y;
END;
$$;


select addition(10,20)
